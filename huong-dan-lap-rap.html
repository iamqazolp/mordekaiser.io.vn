Mechanical & Calibration Rules

Quy tắc cơ khí & hiệu chỉnh thực tế

1. Góc & Gate Radar HLK-LD2410C (Chống nhận nhầm TC03) Gắn dưới gầm bàn, mặt anten patch hướng chếch xuống ngực/bụng người ngồi (~35°). Dùng app HLKRadarTool (Bluetooth) cài Max Distance Gate = Gate 1 (0.75 m) và hạ độ nhạy các gate xa về 0. Khoảng cách 2 radar ≥ 60 cm (nếu làm sa bàn thu nhỏ, lót vách lá đồng/giấy bạc chắn sóng giữa 2 khoang).
2. Cách lót Cảm biến lực FSR402 dưới đệm ghế (TC02) Không dán FSR402 lên bề mặt gồ ghề hay gập cổ cáp! Dán mặt cảm biến lên tấm lót phẳng, đặt 1 miếng mút EVA / cao su tròn đường kính ~1.5 cm ngay tâm vùng tròn rồi mới phủ đệm mỏng lên trên để tập trung trọng lực balo/người ngồi.
3. Vị trí Cụm Môi trường (Zone Node)
  • BH1750: Đặt ngửa hướng lên trần đèn (không giấu dưới gầm bàn vì ngưỡng cảnh báo là < 300 lux tại mặt bàn đọc).
  • INMP441: Khoan lỗ thoáng Ø 3 mm đúng vị trí lỗ thu âm trên nắp hộp để đo chính xác mức dB (cảnh báo > 55 dB).
  • SHT31: Đặt cách chip ESP32 tối thiểu 4 cm để tránh sai số +2°C do nhiệt tỏa từ Wi-Fi.
4. Cổng Ra Vào RFID RC522 (Gate Node) Không gắn sát RC522 lên bề mặt kim loại (nhôm/sắt sẽ triệt tiêu từ trường 13.56 MHz). Chỉ gắn sau tấm Mica hoặc Gỗ dày ≤ 5 mm.

Sketch 01 · Seat + Zone Node (ESP32-S3)

              
Test ESP32-S3 #1 (Bàn 2 chỗ + Môi trường)

            

            
          

          

            Đọc đồng thời 2 FSR402 (GPIO 4/5 ADC1), 2 Radar UART 256000 baud (GPIO 17/18 & 15/16), quét I2C (GPIO 8/9: SHT31 0x44 & BH1750 0x23) và test đổi màu LED WS2812B (GPIO 6/7).
          

          
#include <Arduino.h>
#include <Wire.h>
#include <Adafruit_NeoPixel.h>

// ==== CHÂN CHUẨN ESP32-S3 (TRÁNH PSRAM 26-37 & USB 19-20) ====
#define FSR1_PIN      4    // ADC1_CH3 (Ghế S01)
#define FSR2_PIN      5    // ADC1_CH4 (Ghế S02)
#define RADAR1_RX     17   // Nối vào TX của LD2410C #1
#define RADAR1_TX     18   // Nối vào RX của LD2410C #1
#define RADAR2_RX     15   // Nối vào TX của LD2410C #2
#define RADAR2_TX     16   // Nối vào RX của LD2410C #2
#define LED1_PIN      6    // Dải WS2812B #1 (qua trở 330 ohm)
#define LED2_PIN      7    // Dải WS2812B #2 (qua trở 330 ohm)
#define I2C_SDA       8
#define I2C_SCL       9

Adafruit_NeoPixel ledS01(8, LED1_PIN, NEO_GRB + NEO_KHZ800);
Adafruit_NeoPixel ledS02(8, LED2_PIN, NEO_GRB + NEO_KHZ800);

void scanI2C() {
  Serial.println("[I2C] Quet thiet bi tren GPIO 8/9...");
  for (byte addr = 1; addr < 127; addr++) {
    Wire.beginTransmission(addr);
    if (Wire.endTransmission() == 0) {
      Serial.printf("  -> 0x%02X ", addr);
      if (addr == 0x44 || addr == 0x45) Serial.println("(SHT31 OK)");
      else if (addr == 0x23 || addr == 0x5C) Serial.println("(BH1750 OK)");
      else Serial.println();
    }
  }
}

void setStripColor(Adafruit_NeoPixel &strip, uint32_t color) {
  for (int i = 0; i < strip.numPixels(); i++) strip.setPixelColor(i, color);
  strip.show();
}

void setup() {
  Serial.begin(115200); // Cắm cáp vào cổng UART (hoặc bật USB CDC On Boot nếu cắm cổng USB)
  delay(800);
  analogReadResolution(12);
  Wire.begin(I2C_SDA, I2C_SCL);
  scanI2C();

  Serial2.begin(256000, SERIAL_8N1, RADAR1_RX, RADAR1_TX);
  Serial1.begin(256000, SERIAL_8N1, RADAR2_RX, RADAR2_TX);

  ledS01.begin(); ledS01.setBrightness(60);
  ledS02.begin(); ledS02.setBrightness(60);
}

void loop() {
  int fsr1 = analogRead(FSR1_PIN);
  int fsr2 = analogRead(FSR2_PIN);
  int r1Bytes = Serial2.available();
  int r2Bytes = Serial1.available();

  setStripColor(ledS01, fsr1 > 800 ? ledS01.Color(214, 60, 28) : ledS01.Color(46, 122, 78));
  setStripColor(ledS02, fsr2 > 800 ? ledS02.Color(214, 60, 28) : ledS02.Color(46, 122, 78));

  Serial.printf("[S01] FSR=%4d | R1_Bytes=%3d  ||  [S02] FSR=%4d | R2_Bytes=%3d\n",
                fsr1, r1Bytes, fsr2, r2Bytes);

  while (Serial2.available()) Serial2.read();
  while (Serial1.available()) Serial1.read();
  delay(500);
}

        


        

          

            

              
Sketch 02 · Gate Node (ESP32-S3)

              
Test ESP32-S3 #2 (RFID RC522 + OLED)

            

            
          

          

            Kiểm tra giao tiếp FSPI của RC522 (3.3V, GPIO 10..14), OLED SSD1306 (GPIO 8/9, 0x3C), Buzzer và nút chuyển chiều VÀO / RA.
          

          
#include <Arduino.h>
#include <SPI.h>
#include <Wire.h>
#include <MFRC522.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// ==== CHÂN CHUẨN ESP32-S3 FSPI + I2C ====
#define RC522_SS      10
#define RC522_MOSI    11
#define RC522_SCK     12
#define RC522_MISO    13
#define RC522_RST     14

#define I2C_SDA       8
#define I2C_SCL       9
#define BTN_DIR_PIN   4
#define BUZZER_PIN    5
#define LED_OK_PIN    6
#define LED_ERR_PIN   7

MFRC522 rfid(RC522_SS, RC522_RST);
Adafruit_SSD1306 display(128, 64, &Wire, -1);
bool isDirectionIn = true;

void beep(int ms) {
  digitalWrite(BUZZER_PIN, HIGH); // Đảo LOW/HIGH nếu module còi 3 chân kích mức thấp
  delay(ms);
  digitalWrite(BUZZER_PIN, LOW);
}

void setup() {
  Serial.begin(115200);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_OK_PIN, OUTPUT);
  pinMode(LED_ERR_PIN, OUTPUT);
  pinMode(BTN_DIR_PIN, INPUT_PULLUP);

  Wire.begin(I2C_SDA, I2C_SCL);
  if (display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    display.clearDisplay();
    display.setTextSize(1);
    display.setTextColor(SSD1306_WHITE);
    display.setCursor(0, 0);
    display.println("THU VIEN THONG MINH");
    display.println("Gate Node S3 Ready!");
    display.display();
  }

  SPI.begin(RC522_SCK, RC522_MISO, RC522_MOSI, RC522_SS);
  rfid.PCD_Init();
  Serial.print("RC522 VersionReg: 0x");
  Serial.println(rfid.PCD_ReadRegister(rfid.VersionReg), HEX);
}

void loop() {
  if (digitalRead(BTN_DIR_PIN) == LOW) {
    isDirectionIn = !isDirectionIn;
    beep(60);
    Serial.printf("Chieu cong: %s\n", isDirectionIn ? "VAO (IN)" : "RA (OUT)");
    delay(300);
  }

  if (!rfid.PICC_IsNewCardPresent() || !rfid.PICC_ReadCardSerial()) return;

  String uid = "";
  for (byte i = 0; i < rfid.uid.size; i++) {
    if (rfid.uid.uidByte[i] < 0x10) uid += "0";
    uid += String(rfid.uid.uidByte[i], HEX);
  }
  uid.toUpperCase();

  Serial.printf("[SWIPE %s] UID = %s\n", isDirectionIn ? "IN" : "OUT", uid.c_str());
  digitalWrite(LED_OK_PIN, HIGH);
  beep(120);
  delay(350);
  digitalWrite(LED_OK_PIN, LOW);

  rfid.PICC_HaltA();
  rfid.PCD_StopCrypto1();
}
